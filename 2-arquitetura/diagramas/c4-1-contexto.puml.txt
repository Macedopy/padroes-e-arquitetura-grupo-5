Person(regulador, "Regulador de Leitos", "Operador responsável por gerir a alocação de leitos hospitalares.")

System(sistema_saude, "Sistema Municipal de Saúde", "Gestão de triagem UPA, prontuário eletrônico (PEC), agendamento, vacinação e regulação de leitos.")

System_Ext(cadsus, "CADSUS / e-SUS (Federal)", "Sistema do Ministério da Saúde para validação e sincronização de dados.")
System_Ext(legado_leitos, "Sistema Legado de Leitos", "Sistema legado mantido em operação para regulação de leitos hospitalares.")

Rel(cidadao, sistema_saude, "Consulta agendamentos e vacinas via PWA/Web", "HTTPS")
Rel(profissional, sistema_saude, "Realiza triagem, consultas e prescrições", "HTTPS")
Rel(regulador, sistema_saude, "Solicita e aprova leitos", "HTTPS")

Rel(sistema_saude, cadsus, "Sincroniza prontuários e cadastros", "REST / SOAP")
Rel(sistema_saude, legado_leitos, "Consulta e reserva leitos em tempo real", "Database Link / REST / ACL")
